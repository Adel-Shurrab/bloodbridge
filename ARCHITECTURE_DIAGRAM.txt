┌──────────────────────────────────────────────────────────────────────────────────┐
│                            DATABASE SCHEMA                                      │
└──────────────────────────────────────────────────────────────────────────────────┘

users
├── id (PK)
├── name
├── email (UNIQUE)
├── phone (UNIQUE)
├── password
├── role: 'donor' | 'organization' | 'admin'
├── is_active
├── email_verified_at
├── phone_verified_at
├── created_at, updated_at, deleted_at

donors (belongs to users)
├── id (PK)
├── user_id (FK → users.id)
├── national_id (UNIQUE)
├── gender: 'male' | 'female'
├── birth_date
├── blood_type (nullable): 'O+' | 'O-' | 'A+' | 'A-' | 'B+' | 'B-' | 'AB+' | 'AB-'
├── verified_blood_type (nullable)
├── lat, lng
├── city
├── points, level, total_donations
├── created_at, updated_at, deleted_at

donor_health_profiles (hasOne relation to donors) ⭐ NEW
├── id (PK)
├── donor_id (FK → donors.id)
├── weight (int, kg)
├── height (int, cm)
├── chronic_disease (boolean)
├── recent_donation (boolean)
├── infection (boolean)
├── is_eligible (boolean) ← AUTO-CALCULATED
├── is_smoker (boolean)
├── has_recent_surgery (boolean)
├── surgery_date (date, nullable)
├── next_eligible_date (date, nullable) ← AUTO-CALCULATED
├── last_donation_date (date, nullable)
├── created_at, updated_at, deleted_at
